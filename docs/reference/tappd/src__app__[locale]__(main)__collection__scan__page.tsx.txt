import { getTranslations } from "next-intl/server";
import { getScanProfiles } from "@/app/actions/scan-profiles";
import ScanCalibration from "@/components/scan/ScanCalibration";
import { Link } from "@/i18n/navigation";

export default async function ScanPage() {
  const [profiles, t] = await Promise.all([getScanProfiles(), getTranslations("collection.scan")]);

  return (
    <div className="mx-auto flex w-full max-w-3xl flex-col gap-6 px-4 py-8">
      <div>
        <Link href="/collection" className="text-xs font-medium text-tappd-accent hover:underline">
          {t("back")}
        </Link>
        <h1 className="mt-2 text-2xl font-bold text-tappd-text">{t("title")}</h1>
        <p className="text-sm text-tappd-text-secondary">{t("description")}</p>
      </div>

      <ScanCalibration profiles={profiles} />
    </div>
  );
}
